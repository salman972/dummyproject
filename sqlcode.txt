CREATE SCHEMA airbnb;
GO


CREATE TABLE airbnb.payments (
    payment_id INT PRIMARY KEY,
    payment_method VARCHAR(20),
    payment_status VARCHAR(20),
    processed_date DATE
);



CREATE TABLE airbnb.bookings (
    booking_id INT PRIMARY KEY,
    user_id INT,
    property_id INT,
    booking_date DATE,
    checkin_date DATE,
    checkout_date DATE,
    booking_status VARCHAR(20),
    total_amount DECIMAL(10,2),
    created_at DATETIME
);


CREATE TABLE airbnb.transactions (
    transaction_id INT PRIMARY KEY,
    booking_id INT,
    payment_id INT,
    transaction_date DATE,
    transaction_type VARCHAR(20), -- DEBIT / REFUND
    amount DECIMAL(10,2),
    CONSTRAINT fk_booking
        FOREIGN KEY (booking_id) REFERENCES airbnb.bookings(booking_id),
    CONSTRAINT fk_payment
        FOREIGN KEY (payment_id) REFERENCES airbnb.payments(payment_id)
);
